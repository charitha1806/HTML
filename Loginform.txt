  <script>
    const togglePwBtn = document.getElementById('togglePw');
    const pwInput = document.getElementById('password');

    togglePwBtn.addEventListener('click', () => {
      const isHidden = pwInput.type === 'password';
      pwInput.type = isHidden ? 'text' : 'password';
      togglePwBtn.textContent = isHidden ? 'Hide' : 'Show';
    });

    function showError(msg) {
      const err = document.getElementById('errorMsg');
      document.getElementById('errorText').textContent = msg;
      err.classList.add('show');
    }

    function clearError() {
      document.getElementById('errorMsg').classList.remove('show');
    }

    document.getElementById('signinBtn').addEventListener('click', () => {
      const email = document.getElementById('email').value.trim();
      const pw = document.getElementById('password').value.trim();

      if (!email) {
        showError('Enter your email or mobile phone number.');
        document.getElementById('email').focus();
        return;
      }

      const emailValid = /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email) || /^\d{10,}$/.test(email);
      if (!emailValid) {
        showError('We cannot find an account with that email address. Enter a valid email address or mobile number.');
        document.getElementById('email').focus();
        return;
      }

      if (!pw) {
        showError('Enter your password.');
        document.getElementById('password').focus();
        return;
      }

      clearError();
      alert('Sign-in submitted!\n\nEmail: ' + email + '\n\n(This is a demo — no real authentication happens here.)');
    });

    document.getElementById('createBtn').addEventListener('click', () => {
      alert('Redirecting to account creation page...\n\n(Demo only)');
    });

    document.getElementById('email').addEventListener('input', clearError);
    document.getElementById('password').addEventListener('input', clearError);

    document.addEventListener('keydown', (e) => {
      if (e.key === 'Enter') document.getElementById('signinBtn').click();
    });
  </script>
